function New-Toast([string]$xml)
{
    $doc = New-Object Windows.Data.Xml.Dom.XmlDocument
    $doc.LoadXml($xml)
    $t = New-Object Windows.UI.Notifications.ToastNotification $doc
    $t.Tag = $tag
    $t.Group = $group
    return $t
}

$script:lastValues = $null
function New-Data([int]$k)
{
    $left = 600 - $k
    $v = New-Object 'System.Collections.Generic.Dictionary[string,string]'
    $v['progressTitle'] = 'Waiting for BrowserAI to be idle'
    $v['progressValue'] = ($k / 600.0).ToString('0.0000', $inv)
    $v['progressValueString'] = ('{0}:{1:00} of 10:00' -f [int][math]::Floor($k / 60), ($k % 60))
    $v['progressStatus'] = ('Installs in {0}:{1:00}' -f [int][math]::Floor($left / 60), ($left % 60))
    $script:lastValues = $v
    return (New-Object Windows.UI.Notifications.NotificationData -ArgumentList $v, ([uint32]($k + 1)))
}

function Wait-Until([long]$ms)
{
    while ($clock.ElapsedMilliseconds -lt $ms) { Start-Sleep -Milliseconds 5 }
}

function History-Count
{
    try { return @($hist.GetHistory($appId)).Count } catch { return "error: $($_.Exception.Message)" }
}

# UIA snapshot of every 'New notification' window: every FlexibleToastView, its rect, runtime id and descendants.
function Snap([string]$label)
{
    $A = [System.Windows.Automation.AutomationElement]
    $views = @()
    $text = New-Object System.Collections.Generic.List[string]
    foreach ($h in [Step0.Mon]::ToastWindows())
    {
        $text.Add("window " + [Step0.Mon]::WindowState($h))
        $w = $A::FromHandle($h)
        $cond = New-Object System.Windows.Automation.PropertyCondition ($A::ClassNameProperty), 'FlexibleToastView'
        foreach ($v in $w.FindAll([System.Windows.Automation.TreeScope]::Descendants, $cond))
        {
            $r = $v.Current.BoundingRectangle
            $rid = ($v.GetRuntimeId() -join '.')
            $text.Add(("  view id='{0}' runtimeId={1} rect=({2},{3}) {4}x{5} name='{6}'" -f $v.Current.AutomationId, $rid, $r.X, $r.Y, $r.Width, $r.Height, $v.Current.Name))
            foreach ($e in $v.FindAll([System.Windows.Automation.TreeScope]::Descendants, [System.Windows.Automation.Condition]::TrueCondition))
            {
                $er = $e.Current.BoundingRectangle
                $text.Add(("    {0} id='{1}' class='{2}' name='{3}' rect=({4},{5}) {6}x{7}" -f $e.Current.ControlType.ProgrammaticName, $e.Current.AutomationId, $e.Current.ClassName, $e.Current.Name, $er.X, $er.Y, $er.Width, $er.Height))
            }
            $views += [pscustomobject]@{ Rect = $r; RuntimeId = $rid; Name = $v.Current.Name }
        }
    }
    Set-Content -Path (Join-Path $here "uia-$label.txt") -Value $text -Encoding UTF8
    L ("uia {0}: {1} toast view(s) {2}" -f $label, $views.Count, (($views | ForEach-Object { "runtimeId=$($_.RuntimeId) rect=$($_.Rect)" }) -join ' | '))
    return $views
}

# Full-region capture to raw\, then a crop of the matching toast view to <label>.png.
function Shot([string]$label, [string]$match)
{
    $bmp = [Step0.Mon]::Grab()
    $at = $clock.ElapsedMilliseconds
    $rawPath = Join-Path $raw "$label-uncropped.png"
    $bmp.Save($rawPath, [System.Drawing.Imaging.ImageFormat]::Png)
    $views = Snap $label
    $v = $views | Where-Object { $_.Name -like "*$match*" } | Select-Object -First 1
    if ($v)
    {
        $reg = [Step0.Mon]::Region
        $x = [int][math]::Round($v.Rect.X) - $reg.X
        $y = [int][math]::Round($v.Rect.Y) - $reg.Y
        $w = [int][math]::Round($v.Rect.Width)
        $h = [int][math]::Round($v.Rect.Height)
        L ("shot {0} captured at {1} ms, toast rect in capture ({2},{3}) {4}x{5}" -f $label, $at, $x, $y, $w, $h)
        Set-Content -Path (Join-Path $raw "$label-rect.txt") -Value "$x $y $w $h"
    }
    else
    {
        L "shot $label captured at $at ms, NO matching toast view for '$match'"
    }
    $bmp.Dispose()
}

    "dry: fg=$([Step0.Mon]::Foreground())"
    "dry: toast windows=$(([Step0.Mon]::ToastWindows() | ForEach-Object { [Step0.Mon]::WindowState($_) }) -join ' | ')"
    [Step0.Mon]::Start()

    [Step0.Mon]::Stop()

{

    $d = New-Data $k
    $a = $clock.ElapsedMilliseconds
    $r = $notifier.Update($d, $tag, $group)
    $z = $clock.ElapsedMilliseconds
    $results["$r"] = 1 + [int]$results["$r"]
    L ("UPDATE k={0} seq={1} result={2} callMs={3} status='{4}' value={5} valueString='{6}'" -f $k, ($k + 1), $r, ($z - $a), $lastValues['progressStatus'], $lastValues['progressValue'], $lastValues['progressValueString'])

    {
        $v = Snap 'roi' | Select-Object -First 1
        if ($v)
        {
            $reg = [Step0.Mon]::Region
            [Step0.Mon]::Roi = New-Object System.Drawing.Rectangle ([int]$v.Rect.X - $reg.X - 4), ([int]$v.Rect.Y - $reg.Y - 4), ([int]$v.Rect.Width + 8), ([int]$v.Rect.Height + 8)
            L "roi set to $([Step0.Mon]::Roi)"
        }
    }
    if ($k -eq 2 -or $k -eq 30 -or $k -eq 60)
    {
        Wait-Until ($T0 + 1000 * $k + 500)
        Shot ("toast1-t{0:00}s" -f $k) 'ready to install'
    }
}
