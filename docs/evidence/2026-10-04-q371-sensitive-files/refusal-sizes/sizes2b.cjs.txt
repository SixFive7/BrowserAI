// Lane q371: the real size of the two refusals that name every tool, read off a
// published BrowserAI.Server.exe over stdio. No browser is started: both refusals
// are made before a session is resolved.
//
// node sizes2b.cjs <exe> <scratch> <report>
'use strict';

const { spawn } = require('child_process');
const fs = require('fs');

const [exe, root, reportPath] = process.argv.slice(2);
fs.mkdirSync(root, { recursive: true });

const report = { exe, startedUtc: new Date().toISOString(), answers: {} };
let child;
let buffer = '';
const pending = new Map();
let nextId = 1;
let stderr = '';

function request(method, params, timeoutMs = 120000) {
  const id = nextId++;
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => { pending.delete(id); reject(new Error(`timeout on ${method}`)); }, timeoutMs);
    pending.set(id, (message) => { clearTimeout(timer); resolve(message); });
    child.stdin.write(JSON.stringify({ jsonrpc: '2.0', id, method, params }) + '\n');
  });
}

function textOf(message) {
  return (message?.result?.content ?? []).filter((block) => block.type === 'text').map((block) => block.text).join('\n');
}

(async () => {
  try {
    child = spawn(exe, [], { cwd: root, stdio: ['pipe', 'pipe', 'pipe'], windowsHide: true });
    report.serverPid = child.pid;
    child.stdout.on('data', (chunk) => {
      buffer += chunk.toString('utf8');
      let at;
      while ((at = buffer.indexOf('\n')) >= 0) {
        const line = buffer.slice(0, at).trim();
        buffer = buffer.slice(at + 1);
        if (!line) continue;
        let message;
        try { message = JSON.parse(line); } catch { continue; }
        if (message.id !== undefined && pending.has(message.id)) {
          pending.get(message.id)(message);
          pending.delete(message.id);
        }
      }
    });
    child.stderr.on('data', (chunk) => { stderr += chunk.toString('utf8'); });

    const init = await request('initialize', { protocolVersion: '2025-06-18', capabilities: {}, clientInfo: { name: 'claude-code', version: '0' } });
    report.serverInfo = init.result?.serverInfo;
    child.stdin.write(JSON.stringify({ jsonrpc: '2.0', method: 'notifications/initialized', params: {} }) + '\n');

    // Q261 b: a call before any tools/list, from a client that calls itself Claude Code.
    const stale = await request('tools/call', { name: 'browser_navigate', arguments: { url: 'data:text/html,x', why: 'measuring the refusal' } });
    report.answers.staleList = { isError: stale.result?.isError === true, length: textOf(stale).length, text: textOf(stale) };

    const list = await request('tools/list', {});
    report.toolCount = (list.result?.tools ?? []).length;

    const missing = await request('tools/call', { name: 'browser_frobnicate', arguments: { why: 'measuring the refusal' } });
    report.answers.noSuchTool = { isError: missing.result?.isError === true, length: textOf(missing).length, text: textOf(missing) };
  } catch (failure) {
    report.failure = String(failure && failure.stack || failure);
  } finally {
    if (child) {
      child.stdin.end();
      const exited = await new Promise((resolve) => {
        const timer = setTimeout(() => resolve(false), 30000);
        child.once('exit', (code) => { clearTimeout(timer); report.exitCode = code; resolve(true); });
      });
      if (!exited) { report.killedAfterTimeout = true; child.kill(); }
    }
    report.stderrTail = stderr.slice(-2000);
    report.endedUtc = new Date().toISOString();
    fs.writeFileSync(reportPath, JSON.stringify(report, null, 2));
  }
})();
