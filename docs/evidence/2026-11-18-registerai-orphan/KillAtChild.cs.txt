// Scratch rig for the maintainer's question on a suspended codex.exe left behind. It
// starts nothing itself: the driver starts RegisterAI and hands its handle here. This
// puts RegisterAI in a job object the rig created, with a completion port, and the
// moment the job reports a new process whose image is the stand-in client, it ends
// RegisterAI by its own handle: the forced order "RegisterAI is ended while it is
// starting a client". Every process of the job is opened as it is created, so what
// is left afterwards can be read by its own handle, never found by name.
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Runtime.InteropServices;
using System.Text;
using System.Threading;

public sealed class KillAtChild : IDisposable
{
    [StructLayout(LayoutKind.Sequential)]
    private struct AssociateCompletionPort
    {
        public IntPtr CompletionKey;
        public IntPtr CompletionPort;
    }

    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
    private static extern IntPtr CreateJobObjectW(IntPtr attributes, string name);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern IntPtr CreateIoCompletionPort(IntPtr file, IntPtr existing, UIntPtr key, uint threads);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool SetInformationJobObject(IntPtr job, int infoClass, ref AssociateCompletionPort info, uint length);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool AssignProcessToJobObject(IntPtr job, IntPtr process);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool TerminateJobObject(IntPtr job, uint exitCode);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool TerminateProcess(IntPtr process, uint exitCode);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool GetQueuedCompletionStatus(IntPtr port, out uint bytes, out UIntPtr key, out IntPtr overlapped, uint milliseconds);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool PostQueuedCompletionStatus(IntPtr port, uint bytes, UIntPtr key, IntPtr overlapped);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern IntPtr OpenProcess(uint access, bool inherit, uint pid);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool GetProcessTimes(IntPtr process, out long creation, out long exit, out long kernel, out long user);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern uint WaitForSingleObject(IntPtr handle, uint milliseconds);

    [DllImport("kernel32.dll", SetLastError = true, CharSet = CharSet.Unicode)]
    private static extern bool QueryFullProcessImageNameW(IntPtr process, uint flags, StringBuilder name, ref uint size);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool CloseHandle(IntPtr handle);

    private const uint QueryLimited = 0x1000;
    private const uint Synchronize = 0x00100000;
    private const uint NewProcess = 6;
    private static readonly UIntPtr StopKey = new UIntPtr(0xDEAD);

    public sealed class Seen
    {
        public uint Pid;
        public string Image = "";
        public double NewAtMs;
        internal IntPtr Handle;
    }

    private readonly IntPtr _job;
    private readonly IntPtr _port;
    private readonly Stopwatch _clock;
    private readonly Thread _thread;
    private readonly string _standIn;
    private IntPtr _launcher;
    private uint _launcherPid;
    public readonly List<Seen> Processes = new List<Seen>();
    public readonly List<string> Events = new List<string>();
    public double KilledAtMs = -1;
    public uint KilledOnPid;
    public bool Armed = true;
    public int DelayMs;

    public KillAtChild(Stopwatch clock, string standInImage)
    {
        _clock = clock;
        _standIn = standInImage;
        _job = CreateJobObjectW(IntPtr.Zero, null);
        if (_job == IntPtr.Zero) throw new System.ComponentModel.Win32Exception();
        _port = CreateIoCompletionPort(new IntPtr(-1), IntPtr.Zero, UIntPtr.Zero, 1);
        if (_port == IntPtr.Zero) throw new System.ComponentModel.Win32Exception();
        var info = new AssociateCompletionPort { CompletionKey = _job, CompletionPort = _port };
        if (!SetInformationJobObject(_job, 7, ref info, (uint)Marshal.SizeOf<AssociateCompletionPort>()))
            throw new System.ComponentModel.Win32Exception();
        _thread = new Thread(Loop) { IsBackground = true, Name = "KillAtChild", Priority = ThreadPriority.Highest };
        _thread.Start();
    }

    public void Adopt(IntPtr processHandle, uint pid)
    {
        _launcher = processHandle;
        _launcherPid = pid;
        if (!AssignProcessToJobObject(_job, processHandle))
            throw new System.ComponentModel.Win32Exception();
    }

    private void Loop()
    {
        while (true)
        {
            uint message;
            UIntPtr key;
            IntPtr overlapped;
            if (!GetQueuedCompletionStatus(_port, out message, out key, out overlapped, 0xFFFFFFFF)) continue;
            if (key == StopKey) return;
            var at = _clock.Elapsed.TotalMilliseconds;
            var pid = unchecked((uint)overlapped.ToInt64());
            if (message != NewProcess)
            {
                lock (Processes) Events.Add(string.Format(System.Globalization.CultureInfo.InvariantCulture, "{0:F2} msg={1} pid={2}", at, message, pid));
                continue;
            }

            var seen = new Seen { Pid = pid, NewAtMs = at };
            seen.Handle = OpenProcess(QueryLimited | Synchronize, false, pid);
            if (seen.Handle != IntPtr.Zero)
            {
                var name = new StringBuilder(1024);
                uint size = (uint)name.Capacity;
                if (QueryFullProcessImageNameW(seen.Handle, 0, name, ref size)) seen.Image = name.ToString();
            }

            // The forced order: the launcher is ended the moment the job reports the client.
            if (Armed && pid != _launcherPid && string.Equals(seen.Image, _standIn, StringComparison.OrdinalIgnoreCase) && KilledAtMs < 0)
            {
                if (DelayMs > 0) Thread.Sleep(DelayMs);
                TerminateProcess(_launcher, 1);
                KilledAtMs = _clock.Elapsed.TotalMilliseconds;
                KilledOnPid = pid;
            }

            lock (Processes)
            {
                Events.Add(string.Format(System.Globalization.CultureInfo.InvariantCulture, "{0:F2} msg=6 pid={1} {2}", at, pid, seen.Image));
                Processes.Add(seen);
            }
        }
    }

    /// <summary>Whether a process of the job is still running, how much CPU it has had, read by the handle opened at its creation.</summary>
    public string Describe(Seen seen)
    {
        if (seen.Handle == IntPtr.Zero) return "no handle";
        var running = WaitForSingleObject(seen.Handle, 0) == 0x102;
        long creation, exit, kernel, user;
        var cpu = GetProcessTimes(seen.Handle, out creation, out exit, out kernel, out user) ? (kernel + user) / 10000.0 : -1;
        return string.Format(System.Globalization.CultureInfo.InvariantCulture, "running={0} cpuMs={1:F1}", running, cpu);
    }

    public bool IsRunning(Seen seen) => seen.Handle != IntPtr.Zero && WaitForSingleObject(seen.Handle, 0) == 0x102;

    public double CpuMs(Seen seen)
    {
        long creation, exit, kernel, user;
        return seen.Handle != IntPtr.Zero && GetProcessTimes(seen.Handle, out creation, out exit, out kernel, out user) ? (kernel + user) / 10000.0 : -1;
    }

    /// <summary>Ends what is left in the job this rig created, and nothing else.</summary>
    public bool EndTheJob() => TerminateJobObject(_job, 1);

    public void Dispose()
    {
        PostQueuedCompletionStatus(_port, 0, StopKey, IntPtr.Zero);
        _thread.Join(5000);
        lock (Processes)
        {
            foreach (var seen in Processes) if (seen.Handle != IntPtr.Zero) CloseHandle(seen.Handle);
        }
        CloseHandle(_port);
        CloseHandle(_job);
    }
}
