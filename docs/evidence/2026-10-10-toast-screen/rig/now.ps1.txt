# Scratch, 2026-10-10: shows the product's own toasts (composed by composer\ at db50964e into now\toasts.json)
# one at a time under Windows PowerShell's app id, each about 8 s, removing the previous first, and saves a
# full-size crop of each banner with an 8 px margin. Windows PowerShell 5.1; the hand is evhand.exe, windowless.
param([int]$HoldMs = 8000, [int]$Margin = 8)
$ErrorActionPreference = 'Stop'
$here = Split-Path -Parent $MyInvocation.MyCommand.Path
$dir = Join-Path $here 'now'
Add-Type -AssemblyName System.Drawing, System.Windows.Forms

using System; using System.Drawing; using System.Drawing.Imaging; using System.Runtime.InteropServices;
public static class Snap
{
    public static int[] Grab(Rectangle r, out Bitmap bmp)
    {
        bmp = new Bitmap(r.Width, r.Height, PixelFormat.Format32bppArgb);
        using (var g = Graphics.FromImage(bmp)) g.CopyFromScreen(r.X, r.Y, 0, 0, r.Size);
        var d = bmp.LockBits(new Rectangle(0, 0, r.Width, r.Height), ImageLockMode.ReadOnly, PixelFormat.Format32bppArgb);
        var px = new int[r.Width * r.Height];
        Marshal.Copy(d.Scan0, px, 0, px.Length);
        bmp.UnlockBits(d);
        return px;
    }
    static bool Differs(int a, int b)
    {
        if (a == b) return false;
        return Math.Abs(((a >> 16) & 255) - ((b >> 16) & 255)) >= 12 || Math.Abs(((a >> 8) & 255) - ((b >> 8) & 255)) >= 12 || Math.Abs((a & 255) - (b & 255)) >= 12;
    }
    // The banner's rows: within columns x0..x1, the rows where at least a fifth of the pixels differ from the baseline.
    public static string Rows(int[] baseline, int[] now, int width, int height, int x0, int x1, int yMax)
    {
        int top = -1, bottom = -1, count = 0;
        for (int y = 0; y < yMax; y++)
        {
            int n = 0;
            for (int x = x0; x <= x1; x++) if (Differs(baseline[y * width + x], now[y * width + x])) n++;
            if (n * 5 >= (x1 - x0 + 1)) { if (top < 0) top = y; bottom = y; count++; }
        }
        return top + "," + bottom + "," + count;
    }
    public static void SaveCrop(Bitmap src, Rectangle r, string path)
    {
        using (var c = src.Clone(r, PixelFormat.Format32bppArgb)) c.Save(path, ImageFormat.Png);
    }
}

$appId = '{1AC14E77-02E7-4E5D-B744-2EB1AE5198B7}\WindowsPowerShell\v1.0\powershell.exe'
$group = 'browserai-tp-now'
$hist = [Windows.UI.Notifications.ToastNotificationManager]::History
$exe = Join-Path $here 'evhand\bin\Release\net10.0-windows10.0.20348.0\evhand.exe'

$region = New-Object System.Drawing.Rectangle ($b.Right - 900), ($b.Bottom - 700), 900, 700
