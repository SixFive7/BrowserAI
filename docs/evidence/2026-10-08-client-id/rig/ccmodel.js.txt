// Q369: a scripted Anthropic Messages API stub, so Claude Code runs with no credential and no
// inference. Derived from .work/client-exit/stub/anthropic-stub.js (stateless) and
// .work/client-behaviour/rig/ccstub.js (request capture). STATELESS policy, per request:
//   1. a real BrowserAI tool is offered and this turn has not called it  -> call it
//   2. update_in_flight is offered and this turn called it fewer than N times -> call it
//   3. otherwise -> the text "stub-turn-finished"
// A "turn" starts at the last user message that carries no tool_result.
// STUB_POLICY_FILE (optional): a JSON file re-read on EVERY request, so one stub process serves a
// whole batch and each run writes its own policy and log location:
//   {"logDir","tag","phPerTurn","delayAfterPhMs","prefix","realTool","directory"}
// env fallbacks: STUB_PORT, STUB_LOGDIR, STUB_TAG, STUB_PH_CALLS_PER_TURN, STUB_DELAY_AFTER_PH_RESULT_MS,
//      STUB_REAL_TOOL, STUB_PREFIX, STUB_DIRECTORY
'use strict';
const http = require('http');
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const PORT = Number(process.env.STUB_PORT || 8787);
const POLICY_FILE = process.env.STUB_POLICY_FILE || '';
function policy() {
  let p = {};
  if (POLICY_FILE) { try { p = JSON.parse(fs.readFileSync(POLICY_FILE, 'utf8')); } catch (e) { p = {}; } }
  const prefix = p.prefix || process.env.STUB_PREFIX || 'mcp__browserai__';
  return {
    logDir: p.logDir || process.env.STUB_LOGDIR || '.',
    tag: p.tag || process.env.STUB_TAG || 'ccmodel',
    phPerTurn: Number(p.phPerTurn ?? process.env.STUB_PH_CALLS_PER_TURN ?? 1),
    delayAfterPhMs: Number(p.delayAfterPhMs ?? process.env.STUB_DELAY_AFTER_PH_RESULT_MS ?? 0),
    prefix,
    realTool: p.realTool || process.env.STUB_REAL_TOOL || `${prefix}browserai_list`,
    directory: p.directory || process.env.STUB_DIRECTORY || 'C:/q369-probe',
    realPerTurn: Number(p.realPerTurn ?? 1),
    delayAfterRealMs: Number(p.delayAfterRealMs ?? 0),
    refreshTool: p.refreshTool || null,
    refreshInput: p.refreshInput || null,
  };
}
let P = policy();
function files() {
  fs.mkdirSync(P.logDir, { recursive: true });
  return { EV: path.join(P.logDir, `${P.tag}.events.log`), REQ: path.join(P.logDir, `${P.tag}.requests.jsonl`), SEEN: path.join(P.logDir, `${P.tag}.seen.jsonl`) };
}
const ev = (s) => fs.appendFileSync(files().EV, `${new Date().toISOString()} ${s}\n`);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
let n = 0;

function sse(res, event, data) { res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`); }
const blocks = (m) => (Array.isArray(m.content) ? m.content : [{ type: 'text', text: String(m.content) }]);
const textOf = (c) => (typeof c === 'string' ? c : Array.isArray(c) ? c.map((b) => (b && b.type === 'text' ? b.text : b && b.type ? `[${b.type}]` : '')).join('') : '');

function analyse(body, phTool) {
  const msgs = Array.isArray(body.messages) ? body.messages : [];
  let turnStart = 0;
  msgs.forEach((m, i) => { if (m.role === 'user' && !blocks(m).some((b) => b && b.type === 'tool_result')) turnStart = i; });
  const idToName = new Map();
  const uses = [];
  msgs.forEach((m, i) => {
    if (m.role !== 'assistant') return;
    for (const b of blocks(m)) if (b && b.type === 'tool_use') { idToName.set(b.id, b.name); if (i > turnStart) uses.push(b.name); }
  });
  const results = [];
  msgs.forEach((m, i) => {
    if (m.role !== 'user') return;
    for (const b of blocks(m)) if (b && b.type === 'tool_result') results.push({ i, inTurn: i > turnStart, tool: idToName.get(b.tool_use_id) || null, is_error: b.is_error === true, text: textOf(b.content) });
  });
  // Claude Code 2.1.288 interleaves role "system" messages; the last one that matters is not one.
  const last = [...msgs].reverse().find((m) => m && m.role !== 'system');
  const lastIsPhResult = !!last && last.role === 'user' && blocks(last).some((b) => b && b.type === 'tool_result' && idToName.get(b.tool_use_id) === phTool);
  return { msgs, turnStart, uses, results, lastIsPhResult };
}

function respond(res, body, kind, toolName, input, id, text) {
  const model = body.model || 'claude-sonnet-5';
  const usage = { input_tokens: 10, output_tokens: 1 };
  if (body.stream === false) {
    const content = kind === 'tool' ? [{ type: 'tool_use', id: `toolu_${id}`, name: toolName, input }] : [{ type: 'text', text }];
    res.writeHead(200, { 'content-type': 'application/json' });
    res.end(JSON.stringify({ id: `msg_${id}`, type: 'message', role: 'assistant', model, content, stop_reason: kind === 'tool' ? 'tool_use' : 'end_turn', stop_sequence: null, usage }));
    return;
  }
  res.writeHead(200, { 'content-type': 'text/event-stream', 'cache-control': 'no-cache', connection: 'keep-alive' });
  sse(res, 'message_start', { type: 'message_start', message: { id: `msg_${id}`, type: 'message', role: 'assistant', model, content: [], stop_reason: null, stop_sequence: null, usage } });
  if (kind === 'tool') {
    sse(res, 'content_block_start', { type: 'content_block_start', index: 0, content_block: { type: 'tool_use', id: `toolu_${id}`, name: toolName, input: {} } });
    sse(res, 'content_block_delta', { type: 'content_block_delta', index: 0, delta: { type: 'input_json_delta', partial_json: JSON.stringify(input) } });
  } else {
    sse(res, 'content_block_start', { type: 'content_block_start', index: 0, content_block: { type: 'text', text: '' } });
    sse(res, 'content_block_delta', { type: 'content_block_delta', index: 0, delta: { type: 'text_delta', text } });
  }
  sse(res, 'content_block_stop', { type: 'content_block_stop', index: 0 });
  sse(res, 'message_delta', { type: 'message_delta', delta: { stop_reason: kind === 'tool' ? 'tool_use' : 'end_turn', stop_sequence: null }, usage: { output_tokens: 5 } });
  sse(res, 'message_stop', { type: 'message_stop' });
  res.end();
}

const server = http.createServer((req, res) => {
  let raw = '';
  req.on('data', (c) => { raw += c; });
  req.on('end', async () => {
    P = policy();
    const url = req.url.split('?')[0];
    if (url.endsWith('/count_tokens')) { ev(`REQ ${req.method} ${req.url} -> count_tokens`); res.writeHead(200, { 'content-type': 'application/json' }); res.end(JSON.stringify({ input_tokens: 100 })); return; }
    if (!url.includes('/v1/messages')) {
      ev(`REQ ${req.method} ${req.url} -> 404`);
      if (req.method === 'HEAD') { res.writeHead(200); res.end(); return; }
      res.writeHead(404, { 'content-type': 'application/json' });
      res.end(JSON.stringify({ type: 'error', error: { type: 'not_found_error', message: `stub has no ${url}` } }));
      return;
    }
    let body = {};
    try { body = JSON.parse(raw); } catch (e) { body = {}; }
    const { REQ, SEEN } = files();
    const PREFIX = P.prefix;
    const PH_TOOL = `${PREFIX}update_in_flight`;
    const REAL_TOOL = P.realTool;
    const idx = ++n;
    const id = `${process.pid}_${idx}_${crypto.randomBytes(4).toString('hex')}`;
    const tools = Array.isArray(body.tools) ? body.tools : [];
    const names = tools.map((t) => t && t.name).filter(Boolean);
    const bai = tools.filter((t) => t && typeof t.name === 'string' && t.name.startsWith(PREFIX));
    const a = analyse(body, PH_TOOL);
    const sys = typeof body.system === 'string' ? body.system : JSON.stringify(body.system || '');
    fs.appendFileSync(REQ, JSON.stringify({ n: idx, at: new Date().toISOString(), url: req.url, body }) + '\n');
    if (a.lastIsPhResult && P.delayAfterPhMs > 0) { ev(`#${idx} delaying ${P.delayAfterPhMs} ms after an update_in_flight result`); await sleep(P.delayAfterPhMs); }
    let kind = 'text';
    let tool = null;
    let input = {};
    let text = 'stub-turn-finished';
    const phThisTurn = a.uses.filter((u) => u === PH_TOOL).length;
    // Optional model-side refresh: when the last BrowserAI result of this turn says the list is a
    // placeholder or stale, and the client offers a refresh tool (policy.refreshTool), call it once.
    const lastBai = a.results.filter((r) => r.inTurn && r.tool && r.tool.startsWith(PREFIX)).pop();
    const staleish = !!lastBai && /has finished|placeholder/.test(lastBai.text);
    const refreshTool = P.refreshTool;
    // Tool search (deferred tools): a BrowserAI name the request mentions but does not define is
    // loaded first with ToolSearch "select:...", once per name, the way the model is told to.
    const mentioned = [...new Set(raw.match(new RegExp(`${PREFIX}[A-Za-z0-9_]+`, 'g')) || [])];
    const searched = [];
    for (const m of a.msgs) if (m.role === 'assistant') for (const b of blocks(m)) if (b && b.type === 'tool_use' && b.name === 'ToolSearch') searched.push(String((b.input && b.input.query) || ''));
    const toLoad = mentioned.filter((x) => !names.includes(x) && !searched.some((q) => q.includes(x)));
    if (names.includes('ToolSearch') && toLoad.length) { kind = 'tool'; tool = 'ToolSearch'; input = { query: `select:${toLoad.join(',')}`, max_results: 10 }; }
    else if (names.includes(REAL_TOOL) && a.uses.filter((u) => u === REAL_TOOL).length < P.realPerTurn) {
      if (a.uses.includes(REAL_TOOL) && P.delayAfterRealMs > 0) { ev(`#${idx} delaying ${P.delayAfterRealMs} ms before the next real call`); await sleep(P.delayAfterRealMs); }
      kind = 'tool'; tool = REAL_TOOL; input = { directory: P.directory };
    }
    else if (refreshTool && names.includes(refreshTool) && staleish && !a.uses.includes(refreshTool)) { kind = 'tool'; tool = refreshTool; input = P.refreshInput || {}; }
    else if (names.includes(PH_TOOL) && phThisTurn < P.phPerTurn) { kind = 'tool'; tool = PH_TOOL; input = {}; }
    else if (!bai.length) text = 'stub-turn-finished (no browserai tool was offered)';
    const seen = {
      n: idx, at: new Date().toISOString(), model: body.model, stream: body.stream, toolCount: names.length,
      browseraiTools: bai.map((t) => ({ name: t.name, description: t.description, input_schema: t.input_schema, defer_loading: t.defer_loading })),
      otherToolNames: names.filter((x) => !x.startsWith(PREFIX)),
      phInstrInSystem: sys.includes('PH-INSTRUCTIONS-PLACEHOLDER'), realInstrInSystem: sys.includes('PH-INSTRUCTIONS-REAL'),
      phInstrAnywhere: raw.includes('PH-INSTRUCTIONS-PLACEHOLDER'), realInstrAnywhere: raw.includes('PH-INSTRUCTIONS-REAL'),
      messages: a.msgs.length, turnStart: a.turnStart, usesThisTurn: a.uses,
      browseraiResults: a.results.filter((r) => r.tool && r.tool.startsWith(PREFIX)),
      otherResults: a.results.filter((r) => !(r.tool && r.tool.startsWith(PREFIX))).map((r) => ({ tool: r.tool, is_error: r.is_error, text: r.text.slice(0, 600) })),
      decision: kind === 'tool' ? `tool_use ${tool}` : `text "${text}"`,
    };
    fs.appendFileSync(SEEN, JSON.stringify(seen) + '\n');
    ev(`#${idx} model=${body.model} tools=${names.length} browserai=[${bai.map((t) => t.name).join(',')}] msgs=${a.msgs.length} usesThisTurn=[${a.uses.join(',')}] -> ${seen.decision}`);
    respond(res, body, kind, tool, input, id, text);
  });
});
server.listen(PORT, '127.0.0.1', () => ev(`LISTENING 127.0.0.1:${PORT} pid=${process.pid} policyFile=${POLICY_FILE}`));
