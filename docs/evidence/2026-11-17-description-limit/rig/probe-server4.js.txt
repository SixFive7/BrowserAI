// Round four, 2026-11-17. Where does the client cut a tool description and the server
// instructions now: at 2,048 as measured on 2026-08-18, or at the 4,096 a later
// changelog names? Each string ends in a marker naming its own length, so a cut shows
// as a missing marker, and each is built to its exact length and checked before it is
// published.
const S = '7C4E';
function exact(label, units) {
  const end = ' MK-END-' + label + '-' + S;
  let text = 'MK-START-' + label + '-' + S + ' ';
  while (text.length < units - end.length) text += 'abcdefghij'.charAt(text.length % 10);
  text = text.slice(0, units - end.length) + end;
  if (text.length !== units) throw new Error(label + ' is ' + text.length + ' units, not ' + units);
  return text;
}
const sizes = [2047, 2048, 2049, 2600, 4095, 4096, 4097, 5000];
const TOOLS = sizes.map((units) => ({
  name: 'probe_d' + units,
  description: exact('D' + units, units),
  inputSchema: { type: 'object', properties: {} },
}));
TOOLS.push({
  name: 'probe_param',
  description: 'Parameter control. MK-PARAM-TOOL-' + S,
  inputSchema: { type: 'object', properties: { p: { type: 'string', description: exact('P20000', 20000) } } },
});
const INSTRUCTIONS = exact('I5000', 5000);
if (process.env.PROBE_DUMP) {
  process.stdout.write(JSON.stringify({ tools: TOOLS.map((t) => [t.name, t.description.length]), instructions: INSTRUCTIONS.length }) + '\n');
  process.exit(0);
}
let buf = '';
process.stdin.setEncoding('utf8');
process.stdin.on('data', (d) => {
  buf += d;
  let i;
  while ((i = buf.indexOf('\n')) >= 0) {
    const line = buf.slice(0, i).trim();
    buf = buf.slice(i + 1);
    if (line) handle(line);
  }
});
const send = (o) => process.stdout.write(JSON.stringify(o) + '\n');
function handle(line) {
  let msg;
  try { msg = JSON.parse(line); } catch { return; }
  if (msg.id === undefined || msg.id === null) return;
  const id = msg.id;
  switch (msg.method) {
    case 'initialize':
      send({ jsonrpc: '2.0', id, result: {
        protocolVersion: (msg.params && msg.params.protocolVersion) || '2025-06-18',
        capabilities: { tools: { listChanged: false } },
        serverInfo: { name: 'truncation-probe4', version: '1.0.0' },
        instructions: INSTRUCTIONS,
      } });
      break;
    case 'tools/list': send({ jsonrpc: '2.0', id, result: { tools: TOOLS } }); break;
    case 'tools/call': send({ jsonrpc: '2.0', id, result: { content: [{ type: 'text', text: 'ok' }], isError: false } }); break;
    case 'ping': send({ jsonrpc: '2.0', id, result: {} }); break;
    case 'resources/list': send({ jsonrpc: '2.0', id, result: { resources: [] } }); break;
    case 'prompts/list': send({ jsonrpc: '2.0', id, result: { prompts: [] } }); break;
    default: send({ jsonrpc: '2.0', id, error: { code: -32601, message: 'Method not found' } });
  }
}
