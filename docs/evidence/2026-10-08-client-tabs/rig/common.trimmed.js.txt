const ALLOW = ['ALLUSERSPROFILE', 'COMPUTERNAME', 'ComSpec', 'CommonProgramFiles', 'CommonProgramFiles(x86)', 'CommonProgramW6432',
  'DriverData', 'HOMEDRIVE', 'HOMEPATH', 'LOGONSERVER', 'NUMBER_OF_PROCESSORS', 'OS', 'Path', 'PATHEXT',
  'PROCESSOR_ARCHITECTURE', 'PROCESSOR_IDENTIFIER', 'PROCESSOR_LEVEL', 'PROCESSOR_REVISION', 'ProgramData',
  'ProgramFiles', 'ProgramFiles(x86)', 'ProgramW6432', 'PUBLIC', 'SESSIONNAME', 'SystemDrive', 'SystemRoot',
  'USERDOMAIN', 'USERDOMAIN_ROAMINGPROFILE', 'USERNAME', 'USERPROFILE', 'windir'];
function allowEnv() {
  const out = {};
  const keys = Object.keys(process.env);
  for (const want of ALLOW) {
    const k = keys.find((x) => x.toLowerCase() === want.toLowerCase());
    if (k !== undefined) out[want] = process.env[k];
  }
  return out;
}

const fwd = (p) => p.replace(/\\/g, '/');
function writeCcSeed(cfg, ver, tag, stray) {
  fs.mkdirSync(cfg, { recursive: true });
  const projects = {};
  // The repository's own .mcp.json sits above every scratch folder and Claude Code finds it by
  // walking up; its four servers are refused here so no run starts them (the shake run did once).
  for (const p of Object.values(PROJ)) projects[fwd(p)] = { hasTrustDialogAccepted: true, hasCompletedProjectOnboarding: true, allowedTools: [], enableAllProjectMcpServers: false, enabledMcpjsonServers: [], disabledMcpjsonServers: ['microsoft-learn', 'context7', 'nuget', 'github'] };
  const seed = {
    hasCompletedOnboarding: true, autoUpdates: false, bypassPermissionsModeAccepted: true,
    customApiKeyResponses: { approved: ['stub-key-not-real'], rejected: [] },
    lastReleaseNotesSeen: ver, projects,
    mcpServers: { idstub: { type: 'stdio', command: RIG, args: ['stub', '--logdir-env', 'IDRIG_LOGDIR', '--logdir', stray, '--tag', tag], env: {} } },
  };
  fs.writeFileSync(path.join(cfg, '.claude.json'), JSON.stringify(seed, null, 1));
  // The same refusal as a user setting, where Claude Code 2.1.29x reads it (the list in
  // .claude.json alone did not stop the shake run's second attempt).
  fs.writeFileSync(path.join(cfg, 'settings.json'), JSON.stringify({ enableAllProjectMcpServers: false, enabledMcpjsonServers: [], disabledMcpjsonServers: ['microsoft-learn', 'context7', 'nuget', 'github'], skipDangerousModePermissionPrompt: true, permissions: { allow: ['mcp__idstub__ping'] } }, null, 1));
}

}
