/// <summary>QPC-based monotonic clock shared by every process of the rig (QPC is system-wide).</summary>
static class Clock
{
    public static readonly double TickMs = 1000.0 / Stopwatch.Frequency;
    public static long Now() => Stopwatch.GetTimestamp();
    public static string Ms(long t) => (t * TickMs).ToString("F3", CultureInfo.InvariantCulture);
    public static string Utc() => DateTime.UtcNow.ToString("O", CultureInfo.InvariantCulture);
    public static string Ft(long ft) => ft == 0 ? "" : DateTime.FromFileTimeUtc(ft).ToString("O", CultureInfo.InvariantCulture);
}

/// <summary>One line per event, flushed with every write so a hard kill loses nothing already written.</summary>
sealed class Log
{
    readonly FileStream _fs;
    readonly object _gate = new();
    public Log(string path)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        _fs = new FileStream(path, FileMode.Append, FileAccess.Write, FileShare.ReadWrite | FileShare.Delete, 1);
    }
    public void W(string role, string ev, string detail)
    {
        long t = Clock.Now();
        var line = $"{Clock.Ms(t)}\t{Clock.Utc()}\t{Environment.ProcessId}\t{role}\t{ev}\t{detail.Replace('\n', ' ').Replace('\r', ' ')}\n";
        var b = Encoding.UTF8.GetBytes(line);
        lock (_gate) { _fs.Write(b, 0, b.Length); _fs.Flush(); }
    }
}

static class Args
{
    public static Dictionary<string, string> Parse(string[] a, int start)
    {
        var d = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        for (int i = start; i < a.Length; i++)
        {
            if (!a[i].StartsWith("--", StringComparison.Ordinal)) continue;
            var k = a[i][2..];
            if (i + 1 < a.Length && !a[i + 1].StartsWith("--", StringComparison.Ordinal)) { d[k] = a[i + 1]; i++; }
            else d[k] = "true";
        }
        return d;
    }
}

    public const uint CREATE_SUSPENDED = 0x4, CREATE_UNICODE_ENVIRONMENT = 0x400, CREATE_NO_WINDOW = 0x08000000;
    public const uint JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE = 0x2000;

    public static int ParentPid()
    {
        var pbi = new PROCESS_BASIC_INFORMATION();
        int st = NtQueryInformationProcess(GetCurrentProcess(), 0, ref pbi, Marshal.SizeOf<PROCESS_BASIC_INFORMATION>(), out _);
        return st == 0 ? (int)pbi.InheritedFromUniqueProcessId : -1;
    }

    public static string ImageOf(SafeProcessHandle h)
    {
        var sb = new StringBuilder(1024); uint n = 1024;
        return QueryFullProcessImageNameW(h, 0, sb, ref n) ? sb.ToString() : $"<image? err {Marshal.GetLastWin32Error()}>";
    }

    public static string ImageOfPid(int pid)
    {
        using var h = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, false, pid);
        return h.IsInvalid ? $"<open failed {Marshal.GetLastWin32Error()}>" : ImageOf(h);
    }

    public static string CommandLineOf(SafeProcessHandle h)
    {
        int len = 65536;
        IntPtr buf = Marshal.AllocHGlobal(len);
        try
        {
            int st = NtQueryInformationProcess(h, 60, buf, len, out _);
            if (st != 0) return $"<cmdline? status 0x{st:X8}>";
            ushort bytes = (ushort)Marshal.ReadInt16(buf);
            IntPtr p = Marshal.ReadIntPtr(buf, IntPtr.Size);
            return p == IntPtr.Zero ? "" : Marshal.PtrToStringUni(p, bytes / 2);
        }
        finally { Marshal.FreeHGlobal(buf); }
    }

    public static long CreationTime(SafeProcessHandle h) => GetProcessTimes(h, out var c, out _, out _, out _) ? c : 0;

    public static string OwnJobInfo()
    {
        bool inJob = false;
        using (var self = new SafeProcessHandle(GetCurrentProcess(), false)) IsProcessInJob(self, IntPtr.Zero, out inJob);
        if (!inJob) return "inJob=False";
        var e = new JOBOBJECT_EXTENDED_LIMIT_INFORMATION();
        string flags = QueryInformationJobObject(IntPtr.Zero, 9, ref e, Marshal.SizeOf<JOBOBJECT_EXTENDED_LIMIT_INFORMATION>(), out _)
            ? $"0x{e.BasicLimitInformation.LimitFlags:X}" : $"<err {Marshal.GetLastWin32Error()}>";
        var buf = new byte[8 + 8 * 4096];
        string pids;
        if (QueryInformationJobObject(IntPtr.Zero, 3, buf, buf.Length, out _))
        {
            uint assigned = BitConverter.ToUInt32(buf, 0), inList = BitConverter.ToUInt32(buf, 4);
            var l = new List<string>();
            for (int i = 0; i < inList && i < 64; i++) l.Add(BitConverter.ToUInt64(buf, 8 + 8 * i).ToString(CultureInfo.InvariantCulture));
            pids = $"assigned={assigned} listed={inList} first64=[{string.Join(",", l)}]";
        }
        else pids = $"<list err {Marshal.GetLastWin32Error()}>";
        return $"inJob=True immediateJobLimitFlags={flags} {pids}";
    }

    public static string ConsoleInfo()
    {
        IntPtr w = GetConsoleWindow();
        var list = new uint[64];
        uint n = GetConsoleProcessList(list, 64);
        return $"consoleWindow=0x{w.ToInt64():X} visible={(w != IntPtr.Zero && IsWindowVisible(w))} consoleProcs=[{string.Join(",", list.Take((int)Math.Min(n, 64)))}]";
    }

    public static string FileTypeOf(int std)
    {
        IntPtr h = GetStdHandle(std);
        if (h == IntPtr.Zero || h == new IntPtr(-1)) return "none";
        return GetFileType(h) switch { 1 => "disk", 2 => "char", 3 => "pipe", 0 => "unknown", var x => x.ToString(CultureInfo.InvariantCulture) };
    }

    public static string CtrlName(uint t) => t switch
    {
        0 => "CTRL_C_EVENT", 1 => "CTRL_BREAK_EVENT", 2 => "CTRL_CLOSE_EVENT", 5 => "CTRL_LOGOFF_EVENT", 6 => "CTRL_SHUTDOWN_EVENT", _ => $"CTRL_{t}"
    };

    /// <summary>An unnamed, non-inheritable job carrying KILL_ON_JOB_CLOSE and nothing else, as BrowserAI's JobObject.CreateKillOnClose.</summary>
    public static IntPtr CreateKillOnCloseJob(out uint flagsReadBack)
    {
        IntPtr job = CreateJobObjectW(IntPtr.Zero, null);
        flagsReadBack = 0;
        if (job == IntPtr.Zero) throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error(), "CreateJobObjectW");
        var e = new JOBOBJECT_EXTENDED_LIMIT_INFORMATION();
        e.BasicLimitInformation.LimitFlags = JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE;
        if (!SetInformationJobObject(job, 9, ref e, Marshal.SizeOf<JOBOBJECT_EXTENDED_LIMIT_INFORMATION>()))
            throw new System.ComponentModel.Win32Exception(Marshal.GetLastWin32Error(), "SetInformationJobObject");
        var r = new JOBOBJECT_EXTENDED_LIMIT_INFORMATION();
        if (QueryInformationJobObject(job, 9, ref r, Marshal.SizeOf<JOBOBJECT_EXTENDED_LIMIT_INFORMATION>(), out _)) flagsReadBack = r.BasicLimitInformation.LimitFlags;
        return job;
    }

    /// <summary>Starts a console child with CREATE_NO_WINDOW, suspended, assigns it to the job, resumes it.</summary>
    public static int StartInJob(IntPtr job, string exe, string commandLine, out string error)
    {
        error = "";
        var si = new STARTUPINFOW { cb = Marshal.SizeOf<STARTUPINFOW>() };
        var cmd = new StringBuilder(commandLine, 32768);
        if (!CreateProcessW(exe, cmd, IntPtr.Zero, IntPtr.Zero, false, CREATE_SUSPENDED | CREATE_NO_WINDOW | CREATE_UNICODE_ENVIRONMENT, IntPtr.Zero, null, ref si, out var pi))
        {
            error = $"CreateProcessW err {Marshal.GetLastWin32Error()}";
            return -1;
        }
        if (!AssignProcessToJobObject(job, pi.hProcess)) error = $"AssignProcessToJobObject err {Marshal.GetLastWin32Error()}";
        ResumeThread(pi.hThread);
        CloseHandle(pi.hThread);
        CloseHandle(pi.hProcess);
        return pi.dwProcessId;
    }

    /// <summary>Opens the parent by pid, checks it was created before us, and logs when it exits.</summary>
    public static void WatchParent(int ppid, Log log, string role)
    {
        var h = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION | SYNCHRONIZE, false, ppid);
        if (h.IsInvalid) { log.W(role, "PARENT_WATCH", $"open failed err={Marshal.GetLastWin32Error()}"); return; }
        long pct = CreationTime(h);
        long mine;
        using (var self = new SafeProcessHandle(GetCurrentProcess(), false)) mine = CreationTime(self);
        if (pct == 0 || pct > mine) { log.W(role, "PARENT_WATCH", $"pid {ppid} created {Clock.Ft(pct)} after me {Clock.Ft(mine)}: not my parent, not watched"); h.Dispose(); return; }
        log.W(role, "PARENT_WATCH", $"pid={ppid} image={ImageOf(h)} created={Clock.Ft(pct)}");
        var t = new Thread(() =>
        {
            WaitForSingleObject(h, INFINITE);
            GetExitCodeProcess(h, out uint code);
            GetProcessTimes(h, out _, out long ex, out _, out _);
            log.W(role, "PARENT_EXIT", $"pid={ppid} code={code} exitUtc={Clock.Ft(ex)}");
        }) { IsBackground = true };
        t.Start();
    }
