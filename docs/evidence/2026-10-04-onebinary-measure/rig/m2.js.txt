function aliveNow() {
  try { return JSON.parse(fs.readFileSync(ALIVE, 'utf8')); } catch (e) { return { utc: null, alive: [] }; }
}
const bg = (a) => a.alive.filter((p) => p.role.startsWith('coordinator') || p.role === 'app' || p.role === 'host' || p.role === 'front');

async function waitNoBackground(limitMs) {
  const t0 = Date.now();
  let a = aliveNow();
  while (bg(a).length > 0 && Date.now() - t0 < limitMs) { await sleep(2000); a = aliveNow(); }
  return { waitedMs: Date.now() - t0, remaining: bg(a).map((p) => `${p.pid}:${p.role}`) };
}

function specFor(kind, run, script) {
  const base = { name: run.split('\\').pop(), dir: run, cwd: PROJ, maxMs: 240000, settleMs: 20000 };
  const policy = { path: POLICY, content: JSON.stringify({ logDir: run, namespace: 'mcp__browserai', script }) };
  if (kind === 'ccP') {
    return Object.assign(base, { exe: CLAUDE, args: ccArgs(run, true, 'Use the browserai tools as instructed.'), env: ccEnv(run, false), stdin: 'closed',
      writeFiles: [{ path: `${run}\\cfg\\.claude.json`, content: seed() }, { path: `${run}\\mcp.json`, content: mcpJson() }, policy] });
  }
  if (kind === 'ccSkill') {
    const status = ctl({ subtype: 'mcp_status' }); const sid = JSON.parse(status).request_id;
    return Object.assign(base, { exe: CLAUDE, args: ccArgs(run, false), env: ccEnv(run, true), stdin: 'pipe',
      writeFiles: [{ path: `${run}\\cfg\\.claude.json`, content: seed() }, { path: `${run}\\mcp.json`, content: mcpJson() }, policy],
      steps: [
        { do: 'send', line: ctl({ subtype: 'initialize' }) }, { do: 'waitFor', pattern: '"type":"control_response"', timeoutMs: 30000 },
        { do: 'mark', text: 'waiting 20 s for the front to reach a host (the front waits up to 15 s, then serves in-process)' }, { do: 'sleep', ms: 20000 },
        { do: 'send', line: status }, { do: 'waitFor', pattern: `"request_id":"${sid}"`, timeoutMs: 30000 },
        { do: 'send', line: userMsg('Use the browserai tools as instructed.') }, { do: 'waitFor', pattern: '"type":"result"', timeoutMs: 120000 },
        { do: 'sleep', ms: 1000 }, { do: 'mark', text: 'end: the client is terminated (as VS Code closing takes it)' }, { do: 'terminate' },
      ] });
  }
  if (kind === 'cxE') {
    return Object.assign(base, { exe: CODEX, args: ['exec', '--json', '--skip-git-repo-check', '--dangerously-bypass-approvals-and-sandbox', '-C', PROJ, 'Use the browserai tools as instructed.'],
      env: cxEnv(run), stdin: 'closed', writeFiles: [{ path: `${run}\\home\\config.toml`, content: codexToml() }, policy] });
  }
  // cxA: codex app-server
  return Object.assign(base, { exe: CODEX, args: ['app-server', '--listen', 'stdio://'], env: cxEnv(run), stdin: 'pipe', autoReplyJsonRpc: true,
    writeFiles: [{ path: `${run}\\home\\config.toml`, content: codexToml() }, policy],
    steps: [
      { do: 'send', line: '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"clientInfo":{"name":"onebinary-probe","title":"onebinary probe","version":"1.0.0"}}}' },
      { do: 'waitFor', pattern: '"id":1\\b', timeoutMs: 30000 },
      { do: 'send', line: JSON.stringify({ jsonrpc: '2.0', id: 2, method: 'thread/start', params: { cwd: PROJ } }) },
      { do: 'waitFor', pattern: '"id":2\\b.*?(?:"thread":\\{"id":"(?<THREAD>[^"]+)"|"threadId":"(?<THREAD>[^"]+)"|"thread_id":"(?<THREAD>[^"]+)")', timeoutMs: 60000 },
      { do: 'waitFor', pattern: '"name":"browserai","status":"(ready|failed)"', timeoutMs: 45000 },
      { do: 'send', line: '{"jsonrpc":"2.0","id":3,"method":"turn/start","params":{"threadId":"$THREAD","input":[{"type":"text","text":"Use the browserai tools as instructed."}]}}' },
      { do: 'waitFor', pattern: 'turn/completed', timeoutMs: 120000 },
      { do: 'sleep', ms: 2000 }, { do: 'mark', text: 'end: close stdin' }, { do: 'closeStdin' },
    ] });
}

function runRig(spec) {
  fs.mkdirSync(spec.dir, { recursive: true });
  const specPath = `${spec.dir}\\spec.json`;
  fs.writeFileSync(specPath, JSON.stringify(spec, null, 1));
  const t0 = new Date().toISOString();
  const r = spawnSync(RIG, ['run', specPath], { windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'], timeout: 400000 });
  fs.writeFileSync(`${spec.dir}\\rig.stdout.txt`, (r.stdout || '') + '\n--- stderr ---\n' + (r.stderr || '') + `\nstatus=${r.status} error=${r.error}`);
  return { startUtc: t0, endUtc: new Date().toISOString(), status: r.status };
}

function seenOf(run) {
  try { return fs.readFileSync(`${run}\\model.seen.jsonl`, 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l)); } catch (e) { return []; }
}
function resultTexts(seen) {
  const out = [];
  for (const s of seen) {
    for (const t of (s.toolResults || [])) out.push({ is_error: t.is_error, head: t.head });
    for (const o of (s.outputs || [])) out.push({ head: o.parts.map((p) => p.head).join(' | ') });
  }
  return out;
}
const offered = (seen, cc) => seen.some((s) => (s.tools || []).some((t) => cc ? String(t).startsWith('mcp__browserai__') : String(t).startsWith('mcp__browserai{')));

(async () => {
  // The page server is a process of its own: runRig waits synchronously, and a server in this
  // process would answer nothing while a client runs (the first attempt, batch m2, did exactly that).
  const page = spawn(NODE, [`${M}\\apistub\\page.js`], { env: Object.assign({}, process.env, { PAGE_PORT: String(PAGEPORT), PAGE_LOG: `${OUT}\\page.log` }), windowsHide: true, stdio: 'ignore' });
  const stubs = [page,
    spawn(NODE, [`${M}\\apistub\\ccstub.js`], { env: Object.assign({}, process.env, { STUB_PORT: String(CCPORT), STUB_POLICY_FILE: POLICY, STUB_LOGDIR: OUT }), windowsHide: true, stdio: 'ignore' }),
    spawn(NODE, [`${M}\\apistub\\cxstub.js`], { env: Object.assign({}, process.env, { STUB_PORT: String(CXPORT), STUB_POLICY_FILE: POLICY, STUB_LOGDIR: OUT }), windowsHide: true, stdio: 'ignore' }),
  ];
  plog(`START root=${root} server=${SERVER} stubs=${stubs.map((s) => s.pid).join(',')} page=${PAGEPORT}`);
  await sleep(1500);
  const exits = exitArg.split(','), reps = Number(repsArg || 3);
  try {
    for (let r = 1; r <= reps; r++) for (const ex of exits) {
      const name = `${ex}-r${r}`;
      const dir = `${OUT}\\${name}`;
      if (fs.existsSync(`${dir}\\summary.json`)) { plog(`SKIP ${name}`); continue; }
      fs.mkdirSync(dir, { recursive: true });
      const sum = { run: name, exit: ex };
      const w = await waitNoBackground(480000);
      sum.waitForNoBackground = w;
      sum.backgroundBeforeA = bg(aliveNow()).map((p) => `${p.pid}:${p.role}`);
      plog(`RUN ${name}: background gone after ${w.waitedMs} ms (remaining: ${w.remaining.join(' ') || 'none'})`);
      const sess = `${OUT}\\sessions\\${name}`;
      const url = `http://127.0.0.1:${PAGEPORT}/p/${name}`;
      const cc = ex.startsWith('cc');
      const kindA = ex === 'ccExit' ? 'ccP' : ex === 'ccKill' ? 'ccSkill' : ex === 'cxExec' ? 'cxE' : 'cxA';
      let a = runRig(specFor(kindA, `${dir}\\A`, scriptA(sess, url, cc)));
      let seenA = seenOf(`${dir}\\A`);
      sum.A = Object.assign(a, { offeredBrowserAI: offered(seenA, cc), toolCalls: seenA.filter((s) => s.answer === 'tool').map((s) => s.tool), results: resultTexts(seenA) });
      plog(`  A ${kindA} status=${a.status} offered=${sum.A.offeredBrowserAI} calls=${sum.A.toolCalls.join(',')}`);
      if (!sum.A.offeredBrowserAI || sum.A.toolCalls.length < 2) {
        // A client that met a cold host without its tools (the known Codex first-turn hazard): once more, warm.
        plog(`  A did not get to open the page; one more A against what is now running`);
        a = runRig(specFor(kindA, `${dir}\\A2`, scriptA(sess, url, cc)));
        seenA = seenOf(`${dir}\\A2`);
        sum.A2 = Object.assign(a, { offeredBrowserAI: offered(seenA, cc), toolCalls: seenA.filter((s) => s.answer === 'tool').map((s) => s.tool), results: resultTexts(seenA) });
        plog(`  A2 ${kindA} status=${a.status} offered=${sum.A2.offeredBrowserAI} calls=${sum.A2.toolCalls.join(',')}`);
      }
      await sleep(6000);
      sum.aliveAfterA = aliveNow();
      plog(`  after A: ${sum.aliveAfterA.alive.map((p) => `${p.pid}:${p.role}`).join(' ')}`);
      const kindB = cc ? 'ccP' : 'cxE';
      const b = runRig(specFor(kindB, `${dir}\\B`, scriptB(sess, cc)));
      const seenB = seenOf(`${dir}\\B`);
      sum.B = Object.assign(b, { offeredBrowserAI: offered(seenB, cc), toolCalls: seenB.filter((s) => s.answer === 'tool').map((s) => s.tool), results: resultTexts(seenB) });
      sum.B.snapshotHasMarker = sum.B.results.some((t) => (t.head || '').includes(`ONEBINARY-PAGE-${name}`));
      plog(`  B ${kindB} status=${b.status} calls=${sum.B.toolCalls.join(',')} marker=${sum.B.snapshotHasMarker}`);
      await sleep(3000);
      sum.aliveAfterB = aliveNow();
      fs.writeFileSync(`${dir}\\summary.json`, JSON.stringify(sum, null, 1));
    }
    const w = await waitNoBackground(480000);
    plog(`END background gone after ${w.waitedMs} ms (remaining: ${w.remaining.join(' ') || 'none'})`);
  } finally {
    for (const s of stubs) { try { s.kill(); } catch (e) { /* gone */ } }
    plog('DONE');
  }
})();
