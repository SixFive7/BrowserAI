using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Text;

namespace ExitRig;

/// <summary>
/// Added 2026-10-04: starts the windowless stub the way a parent with no standard handles would
/// (Velopack's Setup.exe after a non-silent install, a person's double-click, the Task Scheduler),
/// and records what the stub does with no client: how long it lives and its exit code. The stub
/// writes its own report of the handles it found.
/// usage: ExitRig launch --exe <exe> --logdir <dir> --log <file> --mode nullhandles|nohandleflag [--reps 3] [--wait-ms 20000]
///   nullhandles   STARTF_USESTDHANDLES with all three handles NULL, nothing inherited
///   nohandleflag  STARTF_USESHOWWINDOW + SW_SHOWNORMAL and no handle flag, nothing inherited
///                 (the shape of a start that only says how to show the window)
/// The child is in a kill-on-close job this process created and closes at the end, so a child
/// that does not leave on its own is ended by the job, and that is recorded.
/// </summary>
static class Launch
{
    public static int Run(string[] a)
    {
        var o = Args.Parse(a, 1);
        string exe = o["exe"], logDir = o["logdir"], mode = o["mode"];
        int reps = int.Parse(o.GetValueOrDefault("reps", "3"));
        int waitMs = int.Parse(o.GetValueOrDefault("wait-ms", "20000"));
        var log = new Log(o["log"]);
        Directory.CreateDirectory(logDir);
        for (int r = 1; r <= reps; r++)
        {
            IntPtr job = Native.CreateKillOnCloseJob(out _);
            var si = new Native.STARTUPINFOW { cb = Marshal.SizeOf<Native.STARTUPINFOW>() };
            if (mode == "nullhandles") { si.dwFlags = 0x100; si.hStdInput = IntPtr.Zero; si.hStdOutput = IntPtr.Zero; si.hStdError = IntPtr.Zero; }
            else { si.dwFlags = 0x1; si.wShowWindow = 1; }
            var env = new StringBuilder();
            foreach (System.Collections.DictionaryEntry kv in Environment.GetEnvironmentVariables()) env.Append(kv.Key).Append('=').Append(kv.Value).Append('\0');
            env.Append("STUB_LOGDIR=").Append(logDir).Append('\0').Append("STUB_EOF_DELAY_MS=0").Append('\0').Append('\0');
            IntPtr envBlock = Marshal.StringToHGlobalUni(env.ToString());
            long t0 = Clock.Now();
            if (!Native.CreateProcessW(exe, new StringBuilder($"\"{exe}\"", 1024), IntPtr.Zero, IntPtr.Zero, false, Native.CREATE_SUSPENDED | Native.CREATE_UNICODE_ENVIRONMENT, envBlock, null, ref si, out var pi))
            {
                log.W("launch", "FAILED", $"rep={r} mode={mode} err={Marshal.GetLastWin32Error()}");
                Marshal.FreeHGlobal(envBlock); Native.CloseHandle(job); continue;
            }
            Marshal.FreeHGlobal(envBlock);
            Native.AssignProcessToJobObject(job, pi.hProcess);
            Native.ResumeThread(pi.hThread);
            Native.CloseHandle(pi.hThread);
            uint wr = WaitForSingleObject(pi.hProcess, (uint)waitMs);
            double ms = (Clock.Now() - t0) * Clock.TickMs;
            GetExitCodeProcess(pi.hProcess, out uint code);
            log.W("launch", "RESULT", $"rep={r} mode={mode} pid={pi.dwProcessId} exitedOnItsOwn={(wr == 0)} afterMs={ms:F1} exitCode={(wr == 0 ? code.ToString() : "still running at the wait's end, ended by closing the job")}");
            Native.CloseHandle(job);
            WaitForSingleObject(pi.hProcess, 5000);
            Native.CloseHandle(pi.hProcess);
            Thread.Sleep(500);
        }
        return 0;
    }

    [DllImport("kernel32.dll", SetLastError = true)] static extern uint WaitForSingleObject(IntPtr h, uint ms);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool GetExitCodeProcess(IntPtr h, out uint code);
}
