#!/usr/bin/env bash
# The rest of the scheduler half after the first attempt's driver was stopped (its page server ran
# inside the driver and could not answer while a client ran): the runs again as batch m2b against
# the same install and the same root watch, then the uninstall, the after-snapshot and the
# installer lock.
# usage: bash m2-rest.sh <reps> <exits>
set -u
M=/c/Source/SixFive7/BrowserAI/.work/onebinary-measure
WT=/c/Source/SixFive7/BrowserAI/.work/wt/onebinary
REPS=${1:-3}
EXITS=${2:-ccExit,ccKill,cxExec,cxApp}
LOG=$M/logs/m2-all.log
export GIT_TERMINAL_PROMPT=0 GCM_INTERACTIVE=never DOTNET_CLI_TELEMETRY_OPTOUT=1
say() { echo "$(date -u +%Y-%m-%dT%H:%M:%SZ) $*" >> "$LOG"; }
WINPID=$(cat /proc/$$/winpid)
say "REST BEGIN reps=$REPS exits=$EXITS bash winpid=$WINPID"
TOKEN=$(pwsh -NoProfile -File "$WT/build/InstallerLock.ps1" -Take -HolderPid "$WINPID" 2>>"$LOG")
say "installer lock: token=$TOKEN rc=$?"
ROOT=$(python -c "import json;print(json.load(open(r'$(cygpath -w $M/m2-state.json)'))['installRoot'])")
WATCH=$M/runs/m2-watch
M2_WATCH="$(cygpath -w $WATCH)" node "$M/rig/m2.js" m2b "$ROOT" "$EXITS" "$REPS" >> "$LOG" 2>&1
say "m2.js (m2b) finished rc=$?"
pwsh -NoProfile -File "$M/rig/m2-snap.ps1" -Tag before-uninstall -InstallRoot "$ROOT" >> "$LOG" 2>&1
pwsh -NoProfile -File "$M/rig/m2-uninstall.ps1" >> "$LOG" 2>&1
say "uninstall script finished"
sleep 5
touch "$WATCH/STOP"
sleep 3
pwsh -NoProfile -File "$M/rig/m2-snap.ps1" -Tag after -InstallRoot "$ROOT" >> "$LOG" 2>&1
pwsh -NoProfile -File "$WT/build/InstallerLock.ps1" -Release -HolderPid "$WINPID" >> "$LOG" 2>&1
say "installer lock released rc=$?"
say "END"
