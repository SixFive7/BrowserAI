import subprocess, json, os, sys, time, hashlib
exe=sys.argv[1]; d=sys.argv[2]
os.makedirs(d, exist_ok=True)
env=dict(os.environ); env['STUB_LOGDIR']=d; env['STUB_EOF_DELAY_MS']='200'
CREATE_NO_WINDOW=0x08000000
p=subprocess.Popen([exe], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env, creationflags=CREATE_NO_WINDOW)
def send(o): p.stdin.write((json.dumps(o)+'\n').encode()); p.stdin.flush()
def recv(): return json.loads(p.stdout.readline())
send({"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-06-18","capabilities":{},"clientInfo":{"name":"smoke","version":"1"}}})
r=recv(); print('init', r['result']['serverInfo'], r['result']['protocolVersion'], list(r['result']['capabilities'].keys()))
send({"jsonrpc":"2.0","method":"notifications/initialized"})
send({"jsonrpc":"2.0","id":2,"method":"tools/list"}); r=recv(); print('tools', [t['name'] for t in r['result']['tools']])
send({"jsonrpc":"2.0","id":3,"method":"tools/call","params":{"name":"big","arguments":{}}})
line=p.stdout.readline(); r=json.loads(line); t=r['result']['content'][0]['text']
print('big frame bytes', len(line), 'text utf8', len(t.encode()), hashlib.sha256(t.encode()).hexdigest())
p.stdin.close(); t0=time.time(); rc=p.wait(); print('exit', rc, 'after', round(time.time()-t0,3))
err=p.stderr.read().decode('utf-8','replace'); print('stderr lines', len(err.splitlines())); print('\n'.join([l for l in err.splitlines() if 'MARKER' in l][:6]))
