// Anthropic Messages API stub for the one-binary measurement (2026-10-04), derived from
// .work\client-exit\stub\anthropic-stub.js (copied). STATELESS: a request offering the tool the
// policy names (default mcp__probe__big), with no tool_result anywhere in the conversation, gets
// a tool_use for it; every other request gets "stub-turn-finished" and end_turn.
// policy.script (optional): a list of {tool, input} steps; step k is answered when the
// conversation already holds k tool_results. Used by the scheduler half.
// Every request is digested into <policy.logDir>\model.seen.jsonl: the tools offered, what the
// stub answered, and every tool_result's text length and sha256.
const http = require('http');
const fs = require('fs');
const crypto = require('crypto');
const PORT = Number(process.env.STUB_PORT || 8971);
const POLICY = process.env.STUB_POLICY_FILE;
const FALLBACK = process.env.STUB_LOGDIR || __dirname;
let n = 0;
const policy = () => { try { return JSON.parse(fs.readFileSync(POLICY, 'utf8')); } catch (e) { return { logDir: FALLBACK }; } };
const sse = (res, event, data) => res.write(`event: ${event}\ndata: ${JSON.stringify(data)}\n\n`);
const sha = (s) => crypto.createHash('sha256').update(Buffer.from(s, 'utf8')).digest('hex');

function toolResults(body) {
  const out = [];
  for (const m of (Array.isArray(body.messages) ? body.messages : [])) {
    if (!m || !Array.isArray(m.content)) continue;
    for (const c of m.content) {
      if (!c || c.type !== 'tool_result') continue;
      let parts = [];
      if (typeof c.content === 'string') parts = [c.content];
      else if (Array.isArray(c.content)) parts = c.content.filter((x) => x && x.type === 'text').map((x) => x.text);
      const text = parts.join('');
      out.push({ tool_use_id: c.tool_use_id, is_error: !!c.is_error, parts: parts.length, chars: text.length, utf8: Buffer.byteLength(text, 'utf8'), sha256: sha(text), head: text.slice(0, 2000) });
    }
  }
  return out;
}

function decide(pol, tools, results) {
  if (Array.isArray(pol.script)) {
    const step = pol.script[results.length];
    if (step && tools.includes(step.tool)) return { kind: 'tool', tool: step.tool, input: step.input || {} };
    return { kind: 'text' };
  }
  const want = pol.tool || 'mcp__probe__big';
  return tools.includes(want) && results.length === 0 ? { kind: 'tool', tool: want, input: {} } : { kind: 'text' };
}

const server = http.createServer((req, res) => {
  let raw = '';
  req.on('data', (c) => { raw += c; });
  req.on('end', () => {
    const url = req.url.split('?')[0];
    const pol = policy();
    const logDir = pol.logDir || FALLBACK;
    const ev = (o) => { try { fs.appendFileSync(`${logDir}\\model.seen.jsonl`, JSON.stringify(Object.assign({ t: new Date().toISOString() }, o)) + '\n'); } catch (e) { /* run dir gone */ } };
    if (url.endsWith('/count_tokens')) { res.writeHead(200, { 'content-type': 'application/json' }); res.end(JSON.stringify({ input_tokens: 100 })); return; }
    if (!url.includes('/v1/messages')) {
      ev({ req: `${req.method} ${req.url}`, answer: '404' });
      if (req.method === 'HEAD') { res.writeHead(200); res.end(); return; }
      res.writeHead(404, { 'content-type': 'application/json' }); res.end(JSON.stringify({ type: 'error', error: { type: 'not_found_error', message: 'stub' } })); return;
    }
    let body = {}; try { body = JSON.parse(raw); } catch (e) { body = {}; }
    const id = `${process.pid}_${++n}_${crypto.randomBytes(4).toString('hex')}`;
    const tools = (Array.isArray(body.tools) ? body.tools : []).map((t) => t && t.name).filter(Boolean);
    const results = toolResults(body);
    const d = decide(pol, tools, results);
    ev({ req: `${req.method} ${req.url}`, model: body.model, stream: body.stream, tools, toolResults: results, answer: d.kind, tool: d.tool, input: d.input, id, bodyBytes: raw.length });
    const model = body.model || 'claude-sonnet-5';
    const usage = { input_tokens: 10, output_tokens: 1 };
    const inputJson = JSON.stringify(d.input || {});
    if (body.stream === false) {
      const content = d.kind === 'tool' ? [{ type: 'tool_use', id: `toolu_${id}`, name: d.tool, input: d.input || {} }] : [{ type: 'text', text: 'stub-turn-finished' }];
      res.writeHead(200, { 'content-type': 'application/json' });
      res.end(JSON.stringify({ id: `msg_${id}`, type: 'message', role: 'assistant', model, content, stop_reason: d.kind === 'tool' ? 'tool_use' : 'end_turn', stop_sequence: null, usage }));
      return;
    }
    res.writeHead(200, { 'content-type': 'text/event-stream', 'cache-control': 'no-cache', connection: 'keep-alive' });
    sse(res, 'message_start', { type: 'message_start', message: { id: `msg_${id}`, type: 'message', role: 'assistant', model, content: [], stop_reason: null, stop_sequence: null, usage } });
    if (d.kind === 'tool') {
      sse(res, 'content_block_start', { type: 'content_block_start', index: 0, content_block: { type: 'tool_use', id: `toolu_${id}`, name: d.tool, input: {} } });
      sse(res, 'content_block_delta', { type: 'content_block_delta', index: 0, delta: { type: 'input_json_delta', partial_json: inputJson } });
    } else {
      sse(res, 'content_block_start', { type: 'content_block_start', index: 0, content_block: { type: 'text', text: '' } });
      sse(res, 'content_block_delta', { type: 'content_block_delta', index: 0, delta: { type: 'text_delta', text: 'stub-turn-finished' } });
    }
    sse(res, 'content_block_stop', { type: 'content_block_stop', index: 0 });
    sse(res, 'message_delta', { type: 'message_delta', delta: { stop_reason: d.kind === 'tool' ? 'tool_use' : 'end_turn', stop_sequence: null }, usage: { output_tokens: 5 } });
    sse(res, 'message_stop', { type: 'message_stop' });
    res.end();
  });
});
server.listen(PORT, '127.0.0.1', () => fs.appendFileSync(`${FALLBACK}\\ccstub.listen.log`, `${new Date().toISOString()} LISTENING 127.0.0.1:${PORT} pid=${process.pid}\n`));
